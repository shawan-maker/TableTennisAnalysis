```python
async def incrby(self, key, amount: int = 1):
    """Redis INCRBY — 按指定增量递增。"""
    k = key.encode() if isinstance(key, str) else key
    val = int(self._store.get(k, b"0")) + amount
    self._store[k] = str(val).encode()
    return val

async def publish(self, channel: str, message: str):
    """Redis PUBLISH — 消息发布到频道（测试中为空操作）。"""
    return 0

async def sadd(self, key, *members):
    """Redis SADD — 添加集合成员。"""
    k = key.encode() if isinstance(key, str) else key
    if k not in self._store:
        self._store[k] = set()
    if not isinstance(self._store[k], set):
        self._store[k] = set()
    added = 0
    for member in members:
        m = member.encode() if isinstance(member, str) else member
        if m not in self._store[k]:
            self._store[k].add(m)
            added += 1
    return added
```
